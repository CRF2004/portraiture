下面这套命令不会动系统 Node，也不会写系统目录。

# 1. 设置私有安装根目录
BASE=/mnt/diskb5/rongfeng_private
mkdir -p "$BASE"/{opt,bin,npm-global,npm-cache,claude-config}

# 2. 判断 CPU 架构
case "$(uname -m)" in
  x86_64) NODE_ARCH=x64 ;;
  aarch64|arm64) NODE_ARCH=arm64 ;;
  *) echo "Unsupported arch: $(uname -m)"; exit 1 ;;
esac

# 3. 从国内 npmmirror 下载 Node 22 最新小版本
NODE_VERSION=$(curl -fsSL https://npmmirror.com/mirrors/node/index.tab | awk '$1 ~ /^v22\./ {print $1; exit}')
echo "Using Node: $NODE_VERSION linux-$NODE_ARCH"

cd "$BASE/opt"
curl -fL "https://npmmirror.com/mirrors/node/${NODE_VERSION}/node-${NODE_VERSION}-linux-${NODE_ARCH}.tar.xz" \
  -o "node-${NODE_VERSION}-linux-${NODE_ARCH}.tar.xz"

tar -xJf "node-${NODE_VERSION}-linux-${NODE_ARCH}.tar.xz"
ln -sfn "$BASE/opt/node-${NODE_VERSION}-linux-${NODE_ARCH}" "$BASE/node"


然后写一个专用环境文件：

cat > "$BASE/env_claude.sh" <<EOF
export PATH="$BASE/node/bin:$BASE/npm-global/bin:\$PATH"

export NPM_CONFIG_CACHE="$BASE/npm-cache"
export NPM_CONFIG_REGISTRY="https://registry.npmmirror.com"
export CLAUDE_CONFIG_DIR="$BASE/claude-config"
EOF

source "$BASE/env_claude.sh"

node -v
npm -v
