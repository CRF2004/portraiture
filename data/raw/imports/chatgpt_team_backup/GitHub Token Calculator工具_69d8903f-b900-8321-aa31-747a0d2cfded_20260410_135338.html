# 3. Python 基本用法

## 示例 1：直接算 cost

from tokencost import calculate_cost

cost = calculate_cost(
    model="gpt-4o-mini",
    input_tokens=1200,
    output_tokens=800
)

print(cost)


输出类似：

0.00342  # USD


---

## 示例 2：从真实 chat log 算 token（组合用）

tokencost 不负责 tokenizer，通常要配 tiktoken：

import tiktoken
from tokencost import calculate_cost

enc = tiktoken.encoding_for_model("gpt-4o-mini")

text = "Hello world..."
tokens = len(enc.encode(text))

cost = calculate_cost(
    model="gpt-4o-mini",
    input_tokens=tokens,
    output_tokens=500
)

print(cost)

# 5. 批量 chat logs 计算（你真正会用到的）

## Step 1：自己算 tokens
import tiktoken

enc = tiktoken.get_encoding("o200k_base")

def count_tokens(messages):
    text = "\n".join([m["content"] for m in messages])
    return len(enc.encode(text))


## Step 2：喂给 tokencost

from tokencost import calculate_cost

input_tokens = count_tokens(user_messages)
output_tokens = count_tokens(ai_messages)

cost = calculate_cost(
    model="gpt-4o-mini",
    input_tokens=input_tokens,
    output_tokens=output_tokens
)
