### 🔧 术语预处理函数

import re

def normalize_ecg_term(term: str) -> str:
    term = term.lower()

    # 去掉弱语义词
    term = re.sub(r"\b(with|without|and|or|of)\b", " ", term)

    # ECG 缩写展开
    replacements = {
        r"\bpr\b": "pr interval",
        r"\bqt\b": "qt interval",
        r"\bqrs\b": "qrs complex"
    }

    for k, v in replacements.items():
        term = re.sub(k, v, term)

    # 多空格清理
    term = re.sub(r"\s+", " ", term).strip()
    return term

### 🔍 改进版 UMLS 查询函数

def query_umls_cui(term):
    if pd.isna(term) or not term.strip():
        return None

    norm_term = normalize_ecg_term(term)

    params = {
        "string": norm_term,
        "searchType": "words",
        "pageNumber": 1,
        "apiKey": UMLS_API_KEY
    }

    try:
        resp = requests.get(UMLS_SEARCH_URL, params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()

        results = data.get("result", {}).get("results", [])
        if not results:
            return None

        # 按 score 排序
        results = sorted(
            [r for r in results if r.get("ui") and r.get("ui") != "NONE"],
            key=lambda x: float(x.get("score", 0)),
            reverse=True
        )

        if results:
            return results[0]["ui"]

        return None

    except Exception as e:
        print(f"[WARN] UMLS query failed for term='{term}': {e}")
        return None

import re

def normalize_ecg_term(term: str) -> str:
    """
    ECG-specific lightweight normalization:
    - remove weak context words
    - expand common ECG abbreviations
    """
    term = term.lower()

    # 去掉弱语义词
    term = re.sub(r"\b(with|without|and|or|of)\b", " ", term)

    # ECG 常见缩写展开
    replacements = {
        r"\bpr\b": "pr interval",
        r"\bqt\b": "qt interval",
        r"\bqrs\b": "qrs complex",
        r"\bst\b": "st segment"
    }

    for pattern, repl in replacements.items():
        term = re.sub(pattern, repl, term)

    # 多空格清理
    term = re.sub(r"\s+", " ", term).strip()

    return term

def query_umls_cui(term):

    try:

        data = resp.json()

            r for r in results

            reverse=True
        )

    except Exception as e:
        print(f"[WARN] UMLS query failed for term='{term}': {e}")

UMLS_SEARCH_URL = "https://uts-ws.nlm.nih.gov/rest/search/current"

SOURCE_VOCABS = ["SNOMEDCT_US", "LNC"]
SLEEP_TIME = 0.2


# ================= ECG 术语清洗 =================
def normalize_ecg_term(term: str) -> str:
    term = term.lower().strip()
    term = re.sub(r"\b(with|without|and|or|of)\b", " ", term)

    replacements = {
        r"\bpr\b": "pr interval",
        r"\bqt\b": "qt interval",
        r"\bqrs\b": "qrs complex",
        r"\bst\b": "st segment"
    }

    for k, v in replacements.items():
        term = re.sub(k, v, term)

    term = re.sub(r"\s+", " ", term).strip()
    return term


# ================= UMLS 查询 =================
def query_umls_cui(term: str):
    if not term or not term.strip():
        return None

    norm = normalize_ecg_term(term)

    params = {
        "string": norm,
        "searchType": "words",
        "sabs": ",".join(SOURCE_VOCABS),
        "pageNumber": 1,
        "apiKey": UMLS_API_KEY
    }

    try:
        r = requests.get(UMLS_SEARCH_URL, params=params, timeout=10)
        r.raise_for_status()
        data = r.json()

        results = data.get("result", {}).get("results", [])
        valid = [x for x in results if x.get("ui") and x.get("ui") != "NONE"]

        if not valid:
            return None

        valid.sort(key=lambda x: float(x.get("score", 0)), reverse=True)
        return valid[0]["ui"]

    except Exception:
        return None


# ================= 主流程 =================
def main():
    df = pd.read_csv(INPUT_CSV)

    name_cols = [c for c in df.columns if c.startswith("name")]

    final_cuis = []
    cui_sources = []
    cui_terms = []

    for _, row in tqdm(df.iterrows(), total=len(df), desc="Mapping to UMLS"):
        mapped = False

        # 1️⃣ FullText
        term = row["FullText"]
        cui = query_umls_cui(term)
        if cui:
            final_cuis.append(cui)
            cui_sources.append("FullText")
            cui_terms.append(term)
            mapped = True
        else:
            # 2️⃣ name1, name2, ...
            for col in name_cols:
                name_term = row[col]
                if pd.isna(name_term):
                    continue
                cui = query_umls_cui(name_term)
                if cui:
                    final_cuis.append(cui)
                    cui_sources.append(col)
                    cui_terms.append(name_term)
                    mapped = True
                    break

        if not mapped:
            final_cuis.append(None)
            cui_sources.append(None)
            cui_terms.append(None)

        time.sleep(SLEEP_TIME)

    df["UMLS_CUI"] = final_cuis
    df["CUI_SOURCE"] = cui_sources
    df["CUI_TERM"] = cui_terms

    df.to_csv(OUTPUT_CSV, index=False)
    print(f"[INFO] Saved to {OUTPUT_CSV}")


if __name__ == "__main__":
    main()

## 一、你给的这些失败示例，本质上是哪几类？

你列的例子非常有代表性，基本覆盖了 PTB-XL 映射失败的“天花板”。

### 1️⃣ *with retrograde conduction*

类型：
- 机制性 electrophysiology finding
- 非完整诊断名

Browser 中的实际情况：
- 很少有 exact 的 SNOMED 主概念叫这个
- 常见的是：
- *Retrograde atrioventricular conduction*
- *Retrograde conduction*（作为 finding / phenomenon）

结论：
✅ 可以手动匹配
⚠️ 但需要你接受“语义提升 / 规范化”

推荐映射到：
- *Retrograde atrioventricular conduction*（finding）

> 这是典型的：ECG 现象 → 临床概念提升，非常适合在方法中说明。

### 2️⃣ *sinus/atrial capture*

类型：
- 复合 ECG 现象
- 斜杠表示 *OR / alternative*

Browser 情况：
- 很难有一个单一概念覆盖
- 但有：
- *Sinus capture*
- *Atrial capture*

结论：
⚠️ 不推荐强行映射为一个 CUI

更合理的做法有两种（择一）：

方案 A（推荐）
- 标记为 *ambiguous / composite*
- 不赋 CUI
- 在论文中说明：

方案 B（如果你必须要）

---

### 3️⃣ *Undetermined rhythm*

类型：
- 诊断不确定性陈述
- Meta-level statement（不是生理发现）

Browser 情况：
- UMLS 里通常不是作为标准 finding
- 即便有，也往往是 administrative / qualifier 类

结论（非常重要）：
❌ 不建议手动映射

- 它不表达心电生理事实
- 只是“当前无法判定”
- 在 PTB-XL 官方流程中，这类 statement 通常也会被排除或降权

> 这是一个“label curation 决策点”，而不是映射问题。

✅ 强烈建议手动匹配
