```python
import re

def normalize_ecg_term(term: str) -> str:
    term = term.lower()

    # 去掉弱语义词
    term = re.sub(r"\b(with|without|and|or|of)\b", " ", term)

    # ECG 缩写展开
    replacements = {
        r"\bpr\b": "pr interval",
        r"\bqt\b": "qt interval",
        r"\bqrs\b": "qrs complex"
    }

    for k, v in replacements.items():
        term = re.sub(k, v, term)

    # 多空格清理
    term = re.sub(r"\s+", " ", term).strip()
    return term
```

### 🔍 改进版 UMLS 查询函数

```python
def query_umls_cui(term):
    if pd.isna(term) or not term.strip():
        return None

    norm_term = normalize_ecg_term(term)

    params = {
        "string": norm_term,
        "searchType": "words",
        "pageNumber": 1,
        "apiKey": UMLS_API_KEY
    }

    try:
        resp = requests.get(UMLS_SEARCH_URL, params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()

        results = data.get("result", {}).get("results", [])
        if not results:
            return None

        # 按 score 排序
        results = sorted(
            [r for r in results if r.get("ui") and r.get("ui") != "NONE"],
            key=lambda x: float(x.get("score", 0)),
            reverse=True
        )

        if results:
            return results[0]["ui"]

        return None

    except Exception as e:
        print(f"[WARN] UMLS query failed for term='{term}': {e}")
        return None
```

```python
import re

def normalize_ecg_term(term: str) -> str:
    """
    ECG-specific lightweight normalization:
    - remove weak context words
    - expand common ECG abbreviations
    """

    # 去掉弱语义词

    }

    for pattern, repl in replacements.items():

    # 多空格清理
```

```python
    for k, v in replacements.items():
        term = re.sub(k, v, term)

    term = re.sub(r"\s+", " ", term).strip()
    return term


# ================= UMLS 查询 =================
def query_umls_cui(term: str):
    if not term or not term.strip():
        return None

    norm = normalize_ecg_term(term)

    params = {
        "string": norm,
        "searchType": "words",
        "sabs": ",".join(SOURCE_VOCABS),
        "pageNumber": 1,
        "apiKey": UMLS_API_KEY
    }

    try:
        r = requests.get(UMLS_SEARCH_URL, params=params, timeout=10)
        r.raise_for_status()
        data = r.json()

        results = data.get("result", {}).get("results", [])
        valid = [x for x in results if x.get("ui") and x.get("ui") != "NONE"]

        if not valid:
            return None

        valid.sort(key=lambda x: float(x.get("score", 0)), reverse=True)
        return valid[0]["ui"]

    except Exception:
        return None


# ================= 主流程 =================
def main():
    df = pd.read_csv(INPUT_CSV)

    name_cols = [c for c in df.columns if c.startswith("name")]

    final_cuis = []
    cui_sources = []
    cui_terms = []

    for _, row in tqdm(df.iterrows(), total=len(df), desc="Mapping to UMLS"):
        mapped = False

        # 1️⃣ FullText
        term = row["FullText"]
        cui = query_umls_cui(term)
        if cui:
            final_cuis.append(cui)
            cui_sources.append("FullText")
            cui_terms.append(term)
            mapped = True
        else:
            # 2️⃣ name1, name2, ...
            for col in name_cols:
                name_term = row[col]
                if pd.isna(name_term):
                    continue
                cui = query_umls_cui(name_term)
                if cui:
                    final_cuis.append(cui)
                    cui_sources.append(col)
                    cui_terms.append(name_term)
                    mapped = True
                    break

        if not mapped:
            final_cuis.append(None)
            cui_sources.append(None)
            cui_terms.append(None)

        time.sleep(SLEEP_TIME)

    df["UMLS_CUI"] = final_cuis
    df["CUI_SOURCE"] = cui_sources
    df["CUI_TERM"] = cui_terms

    df.to_csv(OUTPUT_CSV, index=False)
    print(f"[INFO] Saved to {OUTPUT_CSV}")


if __name__ == "__main__":
    main()
```
