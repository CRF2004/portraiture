import pandas as pd
import numpy as np
from sklearn.preprocessing import MultiLabelBinarizer
from sklearn.model_selection import train_test_split
from sklearn.multiclass import OneVsRestClassifier
from xgboost import XGBClassifier

# --- 路径 ----
feat_path  = "/mnt/data/ptb-xl-a-comprehensive-electrocardiographic-feature-dataset-1.0.1/features/12sl_features.csv"
label_path = "/mnt/data/ptb-xl-a-comprehensive-electrocardiographic-feature-dataset-1.0.1/labels/ptbxl_statements.csv"


# --- 读取数据 ---
df_feat = pd.read_csv(feat_path)
df_label = pd.read_csv(label_path)

# keep ecg_id
df = df_feat.merge(df_label[['ecg_id','scp_codes_ext']], on='ecg_id', how='inner')


# --- 解析 label 列为 list ---
def parse_codes(s):
    # "[('NORM',100),('SR',100)]" → ['NORM','SR']
    try:
        items = eval(s)   # careful, but csv is trusted here
        return [i[0] for i in items]
    except:
        return []

df['labels'] = df['scp_codes_ext'].apply(parse_codes)


# --- MultiLabel Binarizer ---
mlb = MultiLabelBinarizer()
Y = mlb.fit_transform(df['labels'])

# --- X features (remove ids and strings) ---
drop_cols = ['ecg_id','scp_codes_ext','labels']
feature_cols = [c for c in df.columns if c not in drop_cols]
X = df[feature_cols].astype(float).fillna(0.0)


# --- Train/test split ---
X_train, X_test, y_train, y_test = train_test_split(
    X, Y, test_size=0.2, random_state=42
)

# --- XGBoost (快速训练) ---
model = OneVsRestClassifier(
    XGBClassifier(
        n_estimators=200,
        max_depth=5,
        learning_rate=0.05,
        subsample=0.8,
        colsample_bytree=0.8,
        eval_metric='logloss',
        tree_method="hist"
    )
)

model.fit(X_train, y_train)


# --- Global feature importance ---
# 对于 multi-label，取平均 importance（或最大）
importances = np.zeros(len(feature_cols))
for est in model.estimators_:
    importances += est.feature_importances_

importances = importances / len(model.estimators_)

feat_importance = pd.DataFrame({
    'feature': feature_cols,
    'importance': importances
}).sort_values('importance', ascending=False)


# --- 输出TOP特征 ---
print(feat_importance.head(20))

def feature_selection_pipeline(X, y):
"""特征预处理与筛选流程"""

def feature_selection_pipeline(X, y):
"""特征预处理与筛选流程"""
print("\n" + "=" * 60)
print("4. 开始特征预处理与筛选...")

# 初始特征信息
initial_feature_count = X.shape[1]
print(f" - 初始特征数: {initial_feature_count}")

# 检查缺失值
missing_values = X.isnull().sum()
missing_features = missing_values[missing_values > 0]
if len(missing_features) > 0:
print(f" - 发现 {len(missing_features)} 个特征存在缺失值")
print(f" 缺失值最多的5个特征:")
for feat, count in missing_features.head().items():
print(f" * {feat}: {count} 个缺失值 ({count/len(X)*100:.2f}%)")
# 简单填充缺失值
X = X.fillna(X.median())
print(f" - 已用中位数填充缺失值")

# 移除低方差特征
print(f"\n - 正在进行方差筛选 (阈值={VAR_THRESHOLD})...")
selector = VarianceThreshold(threshold=VAR_THRESHOLD)
X_reduced = selector.fit_transform(X)
selected_features_mask = selector.get_support()
feature_names = X.columns[selected_features_mask]

X_df = pd.DataFrame(X_reduced, columns=feature_names, index=X.index)
after_variance_count = X_df.shape[1]
removed_by_variance = initial_feature_count - after_variance_count

print(f" - 方差筛选结果:")
print(f" * 保留特征数: {after_variance_count}")
print(f" * 移除特征数: {removed_by_variance} ({removed_by_variance/initial_feature_count*100:.1f}%)")

# 使用 XGBoost 进行特征重要性评估
print(f"\n - 正在训练 XGBoost 模型评估特征重要性...")
print(f" * 目标标签数: {y.shape[1]}")
print(f" * 训练样本数: {X_df.shape[0]}")

start_time = time.time()

# 创建XGBoost模型
clf = MultiOutputClassifier(
XGBClassifier(
n_estimators=100,
max_depth=5,
learning_rate=0.1,
tree_method='hist',
n_jobs=-1,
random_state=RANDOM_STATE,
verbosity=0 # 减少输出噪音
)
)

# 训练模型并输出进度
print(f" * 开始训练 (这可能需要几分钟)...")

clf.fit(X_df, y)

training_time = time.time() - start_time
print(f" * 训练完成，耗时: {training_time:.1f}秒")

# 聚合特征重要性

importances = np.zeros(len(feature_names))

for i, estimator in enumerate(clf.estimators_):
importances += estimator.feature_importances_
if (i + 1) % 5 == 0 or i == len(clf.estimators_) - 1:
print(f" * 已处理 {i + 1}/{len(clf.estimators_)} 个标签")

feat_imp_df = pd.DataFrame({

}).sort_values(by='Importance', ascending=False)
