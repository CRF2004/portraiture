## 2️⃣ 疾病名归一化（只用于候选发现）

SUFFIX_PATTERNS = [
    "疾病", "病", "综合征", "综合症", "症",

def normalize_disease_name(name: str) -> str:
    name = name.strip()
    for s in SUFFIX_PATTERNS:
        name = re.sub(f"{s}$", "", name)
    name = re.sub(r"[，,。；;（）()]", "", name)

def flatten_relations(data):

## 4️⃣ RapidFuzz 构建相似疾病图

def build_similarity_graph(disease_names, threshold=80):
    G = nx.Graph()
    names = list(disease_names)

    for name in names:
        G.add_node(name)

    normalized = {n: normalize_disease_name(n) for n in names}

    for i in range(len(names)):
        for j in range(i + 1, len(names)):
            a, b = names[i], names[j]

            # blocking：归一化后必须有较长公共部分
            if len(set(normalized[a]) & set(normalized[b])) < 2:
                continue

            score = max(
                fuzz.token_set_ratio(normalized[a], normalized[b]),
                fuzz.partial_ratio(normalized[a], normalized[b])
            )

            if score >= threshold:
                G.add_edge(a, b)

    return G

## 6️⃣ 主流程：删除 + 重建 JSON

def clean_disease_json(input_path, output_path):
    with open(input_path, "r", encoding="utf-8") as f:
        raw_data = json.load(f)

    records = flatten_relations(raw_data)

    disease_desc_map = {}
    for r in records:
        disease_desc_map.setdefault(r["name"], r["description"])

    all_disease_names = set(disease_desc_map.keys())

    # 1. 构图
    G = build_similarity_graph(all_disease_names)

    # 2. 聚类
    clusters = [list(c) for c in nx.connected_components(G) if len(c) > 1]

    to_remove = set()

    for cluster in clusters:
        decision = llm_decide_cluster(cluster, disease_desc_map)
        if decision:
            to_remove.update(decision["remove"])

    # 3. 删除疾病关系
    new_data = []
    for item in raw_data:
        new_diseases = [
            d for d in item["diseases"]
            if d["name"] not in to_remove
        ]
        if new_diseases:
            new_data.append({
                "indicatorName": item["indicatorName"],
                "diseases": new_diseases
            })

    # 4. 保存
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(new_data, f, ensure_ascii=False, indent=2)

    print(f"完成：删除疾病数 = {len(to_remove)}，输出至 {output_path}")

### 3️⃣ 高性能相似图构建（完全替换原函数）

from rapidfuzz import fuzz
import networkx as nx

def build_similarity_graph_fast(disease_names, sim_threshold=80, min_token_overlap=2):
    G = nx.Graph()
    for name in disease_names:
        G.add_node(name)

    token_index, disease_tokens = build_token_index(disease_names)

    checked_pairs = set()

    for name in disease_names:
        candidates = set()
        for t in disease_tokens[name]:
            candidates |= token_index[t]

        for other in candidates:
            if other <= name:
                continue

            key = (name, other)
            if key in checked_pairs:
                continue
            checked_pairs.add(key)

            # token overlap 再过滤一次
            if len(disease_tokens[name] & disease_tokens[other]) < min_token_overlap:
                continue

            score = max(
                fuzz.token_set_ratio(name, other),
                fuzz.partial_ratio(name, other)
            )

            if score >= sim_threshold:
                G.add_edge(name, other)

    return G

def normalize_llm_decision(decision: dict):
    """
    强制将 LLM decision 规范化为：
    {
      keep: str,
      remove: List[str]
    }
    """
    if decision is None:
        return None

    keep = decision.get("keep")
    remove = decision.get("remove", [])

    # keep 是 list 的情况（常见）
    if isinstance(keep, list):
        if len(keep) == 1:
            keep = keep[0]
        else:
            # 多个 keep，取“最长的”（保守策略）
            keep = max(keep, key=len)

    # keep 不是字符串，直接失败
    if not isinstance(keep, str):
        return None

    # remove 不是 list 的情况
    if isinstance(remove, str):
        remove = [remove]
    elif not isinstance(remove, list):
        remove = []

    # 移除 keep 自身
    remove = [r for r in remove if r != keep]

    return {
        "keep": keep,
        "remove": remove,
        "reason": decision.get("reason", "")
    }
