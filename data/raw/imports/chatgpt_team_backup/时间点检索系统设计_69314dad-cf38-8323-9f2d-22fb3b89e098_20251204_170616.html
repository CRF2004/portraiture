## ✅ 检索逻辑

1. 输入：一个 TimePoint JSON（带 bucket 属性）。
2. 候选集生成（可配置 Top-N）：
- 同桶 Top-N
- 父桶 Top-M
- 子桶 Top-L
→ 使用 py2neo 查询 Neo4j，获取 TimePoint 节点及所属患者完整时间线。
3. 相似度计算：
- 对每种事件类型，只要事件匹配就加分（不用事先设权重）：
- VitalState：按 state_summary_* 是否有重复的名称/值计数，不考虑 numeric 差异。
- Lab：按 item 名称重复计数。
- Medication：按 ATC4/药物名称重复计数。
- Procedure：按 procedure_name 重复计数。
- Admission / Discharge：按类型匹配加 1 分。
- 总分 = 所有匹配事件的计数和。
4. 返回：按总分排序 Top-K TimePoint + 患者完整时间线。

> 优点：不需要提前设权重，也不会因为 numeric 差异引入复杂计算；逻辑简单直观。

## ✅ py2neo 查询逻辑

- 根据 TimePoint 的 bucket 查询候选节点。
- 支持父桶、子桶查询。
- MATCH 查询获取 TimePoint 的所有事件和所属患者轨迹。

from py2neo import Graph

class TimePointRetriever:
    def __init__(self, uri, user, password):
        self.graph = Graph(uri, auth=(user, password))

    def get_candidates(self, bucket, top_same=50, top_parent=30, top_child=20, parent_buckets=[], child_buckets=[]):
        # 查询同桶
        same_query = f"""
        MATCH (tp:TimePoint {{bucket: '{bucket}'}})-[:HAS_EVENT]->(e)
        RETURN tp, collect(e) as events
        LIMIT {top_same}
        """
        same_results = self.graph.run(same_query).data()
        
        # 父桶
        parent_results = []
        for pb in parent_buckets:
            q = f"""
            MATCH (tp:TimePoint {{bucket: '{pb}'}})-[:HAS_EVENT]->(e)
            RETURN tp, collect(e) as events
            LIMIT {top_parent}
            """
            parent_results += self.graph.run(q).data()
        
        # 子桶
        child_results = []
        for cb in child_buckets:
            q = f"""
            MATCH (tp:TimePoint {{bucket: '{cb}'}})-[:HAS_EVENT]->(e)
            RETURN tp, collect(e) as events
            LIMIT {top_child}
            """
            child_results += self.graph.run(q).data()
        
        return same_results + parent_results + child_results

    def compute_similarity(self, query_tp, candidate_tp):
        score = 0
        q_events = query_tp['events']
        c_events = candidate_tp['events']

        # 建立 label -> set(mapping key) 的索引
        def event_key(ev):
            if ev['label'] == 'VitalState':
                return tuple(ev.get(k) for k in ev if k.startswith('state_summary_'))
            elif ev['label'] == 'Lab':
                return ev['item']
            elif ev['label'] == 'Medication':
                return ev.get('ATC4', ev.get('drug'))
            elif ev['label'] == 'Procedure':
                return ev['procedure_name']
            else:  # Admission/Discharge
                return ev['label']

        q_keys = set([event_key(ev) for ev in q_events])
        c_keys = set([event_key(ev) for ev in c_events])

        # 简单相交计数
        score = len(q_keys & c_keys)
        return score

    def retrieve_top_k(self, query_tp, k=10, **kwargs):
        candidates = self.get_candidates(query_tp['bucket'], **kwargs)
        scored = []
        for cand in candidates:
            score = self.compute_similarity(query_tp, cand)
            scored.append((score, cand))
        scored.sort(key=lambda x: x[0], reverse=True)
        return scored[:k]

from py2neo import Graph
from typing import List, Dict, Any

class TimePointRetriever:
    def __init__(self, uri: str, user: str, password: str):
        self.graph = Graph(uri, auth=(user, password))

    def get_candidates(self, bucket: str, top_same: int = 50,
                       top_parent: int = 30, top_child: int = 20,
                       parent_buckets: List[str] = [], child_buckets: List[str] = []) -> List[Dict[str, Any]]:
        """
        查询候选 TimePoint
        """
        candidates = []

        # 同桶
        same_query = f"""
        MATCH (tp:TimePoint {{bucket: '{bucket}'}})-[:HAS_EVENT]->(e)
        RETURN tp, collect(e) as events
        LIMIT {top_same}
        """
        candidates += self.graph.run(same_query).data()

        # 父桶
        for pb in parent_buckets:
            q = f"""
            MATCH (tp:TimePoint {{bucket: '{pb}'}})-[:HAS_EVENT]->(e)
            RETURN tp, collect(e) as events
            LIMIT {top_parent}
            """
            candidates += self.graph.run(q).data()

        # 子桶
        for cb in child_buckets:
            q = f"""
            MATCH (tp:TimePoint {{bucket: '{cb}'}})-[:HAS_EVENT]->(e)
            RETURN tp, collect(e) as events
            LIMIT {top_child}
            """
            candidates += self.graph.run(q).data()

        return candidates

    def event_key(self, ev: Dict[str, Any]):
        """
        将事件转换成可比较的 key
        """
        label = ev.get('label')
        if label == 'VitalState':
            # 使用所有 state_summary_* 作为匹配 key
            keys = tuple(ev.get(k) for k in sorted(ev.keys()) if k.startswith('state_summary_'))
            return (label, keys)
        elif label == 'Lab':
            return (label, ev.get('item'))
        elif label == 'Medication':
            # 优先用 ATC4，如果没有再用药名
            return (label, ev.get('ATC4') or ev.get('drug'))
        elif label == 'Procedure':
            return (label, ev.get('procedure_name'))
        else:  # Admission / Discharge
            return (label, )

    def compute_similarity(self, query_tp: Dict[str, Any], candidate_tp: Dict[str, Any]) -> int:
        """
        简单重复事件计数相似度
        """
        q_events = query_tp.get('events', [])
        c_events = candidate_tp.get('events', [])

        q_keys = set([self.event_key(ev) for ev in q_events])
        c_keys = set([self.event_key(ev) for ev in c_events])

        # 相交事件数量即相似度
        score = len(q_keys & c_keys)
        return score

    def get_patient_timeline(self, timepoint_id: int) -> List[Dict[str, Any]]:
        """
        返回某个 TimePoint 所属患者的完整时间线（按 NEXT 链）
        """
        q = f"""
        MATCH (start:TimePoint)-[:NEXT*0..]->(tp:TimePoint)
        WHERE id(start) = {timepoint_id}
        OPTIONAL MATCH (tp)-[:HAS_EVENT]->(e)
        RETURN tp, collect(e) as events
        ORDER BY tp.rel_time_min
        """
        results = self.graph.run(q).data()
        timeline = [{'tp': r['tp'], 'events': r['events']} for r in results]
        return timeline

    def retrieve_top_k(self, query_tp: Dict[str, Any], k: int = 10,
                       top_same: int = 50, top_parent: int = 30, top_child: int = 20,
                       parent_buckets: List[str] = [], child_buckets: List[str] = []) -> List[Dict[str, Any]]:
        """
        输入 TimePoint JSON，返回 Top-K 相似 TimePoint + 患者完整时间线
        """
        candidates = self.get_candidates(
            bucket=query_tp['bucket'],
            top_same=top_same,
            top_parent=top_parent,
            top_child=top_child,
            parent_buckets=parent_buckets,
            child_buckets=child_buckets
        )

        scored = []
        for cand in candidates:
            score = self.compute_similarity(query_tp, cand)
            if score > 0:  # 可选：只返回有匹配的
                # 获取完整患者时间线
                timeline = self.get_patient_timeline(cand['tp'].identity)
                scored.append({'score': score, 'candidate_tp': cand['tp'], 'timeline': timeline})

        # 按分数降序排序
        scored.sort(key=lambda x: x['score'], reverse=True)

        return scored[:k]

        keys = tuple(ev.get(k) for k in sorted(ev.keys()) if k.startswith("state_summary_"))
        return ("VitalState", keys)

        return ("Medication", ev.get("ATC4") or ev.get("drug"))

    elif labels == "Procedure":
        return ("Procedure", ev.get("procedure_name"))

    else:  # Admission / Discharge
        return (labels,)
